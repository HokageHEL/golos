# Golos v1 — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers-extended-cc:subagent-driven-development (recommended) or superpowers-extended-cc:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship a working audio-transcription web app: upload → transcript with speakers and timestamps → edit → export, with an anonymous free trial and a credit ledger ready for billing.

**Architecture:** A single Next.js App Router app on Vercel. No worker, no job queue — the browser uploads directly to Supabase Storage (bypassing Vercel's 4.5 MB body limit) and AssemblyAI's async API holds the long-running work and calls back via webhook. Money is an append-only ledger, never a mutable balance field.

**Tech Stack:** TypeScript, Next.js 15 (App Router), next-intl, Supabase (Postgres + Storage + Auth), AssemblyAI Universal-2, Upstash Redis, Cloudflare Turnstile, Vitest.

**User decisions (already made):**
- Global market, multilingual audio with auto-detection.
- Thin MVP — managed services only, nothing self-hosted.
- Stack is TypeScript / React / Next.js.
- Free trial: one file up to 10 minutes, anonymous, bound to more than IP so a page refresh doesn't reset it.
- v1 includes timestamps (`.srt`) and speaker diarization. No AI summary.
- UI is `en` (primary) + `uk`, architected for more locales later.
- Regional pricing is a later option, not v1.
- Acquiring/legal is the owner's call, outside this plan. Billing is therefore last and isolated.

**Spec:** `docs/superpowers/specs/2026-08-08-golos-design.md`

```ts
import { expect, test } from 'vitest';
import en from '../../messages/en.json';
import uk from '../../messages/uk.json';

function flatten(obj: Record<string, unknown>, prefix = ''): string[] {
  return Object.entries(obj).flatMap(([k, v]) =>
    typeof v === 'object' && v !== null
      ? flatten(v as Record<string, unknown>, `${prefix}${k}.`)
      : [`${prefix}${k}`],
  );
}

test('en and uk have identical key sets', () => {
  expect(flatten(uk).sort()).toEqual(flatten(en).sort());
});
```

```sql
create type job_status as enum ('awaiting_upload', 'processing', 'done', 'failed');

create table accounts (
  user_id uuid primary key references auth.users on delete cascade,
  balance_cents integer not null default 0,
  plan text not null default 'payg',
  subscription_status text,
  minutes_included integer not null default 0,
  minutes_used integer not null default 0,
  period_start timestamptz,
  created_at timestamptz not null default now()
);

create table jobs (
  id uuid primary key default gen_random_uuid(),
  user_id uuid references auth.users on delete cascade,
  anon_key text,
  filename text not null,
  storage_path text not null,
  status job_status not null default 'awaiting_upload',
  provider text not null default 'assemblyai',
  provider_job_id text unique,
  duration_seconds integer,
  language text,
  cost_cents integer,
  truncated boolean not null default false,
  error text,
  created_at timestamptz not null default now(),
  completed_at timestamptz,
  constraint jobs_owner_present check (user_id is not null or anon_key is not null)
);

create index jobs_user_created_idx on jobs (user_id, created_at desc);

create table transcripts (
  job_id uuid primary key references jobs on delete cascade,
  text text not null,
  segments jsonb not null,
  word_count integer not null,
  created_at timestamptz not null default now()
);

-- Append-only. Never UPDATE or DELETE rows here.
create table ledger (
  id bigserial primary key,
  user_id uuid not null references auth.users on delete cascade,
  delta_cents integer not null,
  reason text not null,
  job_id uuid references jobs on delete set null,
  external_id text not null unique,
  created_at timestamptz not null default now()
);

create index ledger_user_idx on ledger (user_id);

-- balance_cents is a cache of sum(ledger.delta_cents), maintained here so that
-- no application code is ever trusted to keep it in sync.
create function refresh_balance() returns trigger language plpgsql as $$
begin
  update accounts
     set balance_cents = (select coalesce(sum(delta_cents), 0)
                            from ledger where user_id = new.user_id)
   where user_id = new.user_id;
  return new;
end;
$$;

create trigger ledger_refresh_balance
  after insert on ledger
  for each row execute function refresh_balance();

alter table accounts    enable row level security;
alter table jobs        enable row level security;
alter table transcripts enable row level security;
alter table ledger      enable row level security;

create policy own_account on accounts
  for select using (auth.uid() = user_id);

create policy own_jobs on jobs
  for select using (auth.uid() = user_id);

create policy own_transcripts on transcripts
  for select using (
    exists (select 1 from jobs j where j.id = job_id and j.user_id = auth.uid())
  );

create policy own_ledger on ledger
  for select using (auth.uid() = user_id);
```

Anonymous jobs have `user_id is null`, so no RLS policy matches them — they are unreachable by the public client by design. The anonymous result page reads its job through the service-role client, gated on the job id (a UUID) plus the `anon_key` cookie.

```ts
export function billableMinutes(durationSeconds: number): number {
  if (durationSeconds <= 0) return 0;
  return Math.ceil(durationSeconds / 60);
}

export type PlanState = { minutesIncluded: number; minutesUsed: number };

export type Charge = { minutesFromPlan: number; centsFromCredit: number };

export function chargeFor(durationSeconds: number, plan: PlanState): Charge {
  const minutes = billableMinutes(durationSeconds);
  const remaining = Math.max(0, plan.minutesIncluded - plan.minutesUsed);
  const minutesFromPlan = Math.min(minutes, remaining);

  return {
    minutesFromPlan,
    centsFromCredit: (minutes - minutesFromPlan) * PAYG_RATE_CENTS,
  };
}
```

```ts
export function toTurns(segments: Segment[]): Turn[] {
  const turns: Turn[] = [];

  for (const s of segments) {
    const last = turns[turns.length - 1];
    if (last && last.speaker === s.speaker) {

      last.endMs = s.endMs;
    } else {
```

```ts
import type { Turn } from '@/lib/transcript/types';

export type SpeakerLabeller = (index: number) => string;

/** Stable 1-based numbering in order of first appearance. */
export function speakerNumbers(turns: Turn[]): Map<string, number> {
  const numbers = new Map<string, number>();
  for (const t of turns) {
    if (t.speaker !== null && !numbers.has(t.speaker)) {
      numbers.set(t.speaker, numbers.size + 1);
    }
  }
  return numbers;
}

export function toPlainText(turns: Turn[], label: SpeakerLabeller): string {
  const numbers = speakerNumbers(turns);
  const showLabels = numbers.size > 1;

  return turns
    .map((t) => {
      if (!showLabels || t.speaker === null) return t.text;
      return `${label(numbers.get(t.speaker)!)}: ${t.text}`;
    })
    .join('\n\n');
}
```

```ts
import type { Turn } from '@/lib/transcript/types';
import { speakerNumbers, type SpeakerLabeller } from './txt';

export async function toDocx(turns: Turn[], label: SpeakerLabeller): Promise<Buffer> {
  const numbers = speakerNumbers(turns);
  const showLabels = numbers.size > 1;

  const paragraphs = turns.map((t) => {
    const runs: TextRun[] = [];
    if (showLabels && t.speaker !== null) {

  });

  return Packer.toBuffer(new Document({ sections: [{ children: paragraphs }] }));
}
```

```ts
import { timingSafeEqual } from 'node:crypto';
import type { Segment } from '@/lib/transcript/types';
import type { SubmitOptions, TranscriptionProvider, TranscriptionResult } from './types';

const API = 'https://api.assemblyai.com/v2/transcript';
export const WEBHOOK_SECRET_HEADER = 'x-golos-webhook-secret';

type Utterance = { start: number; end: number; speaker: string | null; text: string };

export class AssemblyAIProvider implements TranscriptionProvider {
  readonly name = 'assemblyai';

  constructor(private config: { apiKey: string; webhookSecret: string }) {}

  private headers() {
    return { authorization: this.config.apiKey, 'content-type': 'application/json' };
  }

  async submit(options: SubmitOptions): Promise<{ providerJobId: string }> {
    const body: Record<string, unknown> = {
      audio_url: options.audioUrl,
      speaker_labels: true,
      language_detection: true,
      webhook_url: options.webhookUrl,
      webhook_auth_header_name: WEBHOOK_SECRET_HEADER,
      webhook_auth_header_value: this.config.webhookSecret,
    };

    if (options.maxDurationMs !== undefined) body.audio_end_at = options.maxDurationMs;

    const res = await fetch(API, {
      method: 'POST',
      headers: this.headers(),
      body: JSON.stringify(body),
    });

    if (!res.ok) throw new Error(`AssemblyAI submit failed: ${res.status} ${await res.text()}`);

    const json = (await res.json()) as { id: string };
    return { providerJobId: json.id };
  }

  async fetchResult(providerJobId: string): Promise<TranscriptionResult> {
    const res = await fetch(`${API}/${providerJobId}`, { headers: this.headers() });
    if (!res.ok) throw new Error(`AssemblyAI fetch failed: ${res.status}`);

    const json = (await res.json()) as {
      id: string;
      status: string;
      text?: string;
      audio_duration?: number;
      language_code?: string;
      error?: string | null;
      utterances?: Utterance[] | null;
    };

    if (json.status === 'error') {
      return {
        providerJobId: json.id,
        status: 'failed',
        segments: [],
        text: '',
        durationSeconds: 0,
        language: null,
        error: json.error ?? 'Unknown provider error',
      };
    }

    const segments: Segment[] = (json.utterances ?? []).map((u) => ({
      startMs: u.start,
      endMs: u.end,
      speaker: u.speaker,
      text: u.text,
    }));

    return {
      providerJobId: json.id,
      status: 'done',
      segments,
      text: json.text ?? '',
      durationSeconds: json.audio_duration ?? 0,
      language: json.language_code ?? null,
      error: null,
    };
  }

  verifyWebhook(headers: Headers): boolean {
    const provided = headers.get(WEBHOOK_SECRET_HEADER);
    if (provided === null) return false;

    const a = Buffer.from(provided);
    const b = Buffer.from(this.config.webhookSecret);
    return a.length === b.length && timingSafeEqual(a, b);
  }
}
```

The comparison is constant-time and length-checked first, because `timingSafeEqual` throws on mismatched lengths.

```ts
export interface QuotaStore {
  claimOnce(key: string, ttlSeconds: number): Promise<boolean>;
  incrementBy(key: string, amount: number, ttlSeconds: number): Promise<number>;
  get(key: string): Promise<number>;
}

export function redisStore(client: Redis = redis()): QuotaStore {
  return {
    async claimOnce(key, ttlSeconds) {
      const result = await client.set(key, 1, { nx: true, ex: ttlSeconds });
      return result === 'OK';
    },
    async incrementBy(key, amount, ttlSeconds) {
      const value = await client.incrby(key, amount);
      if (value === amount) await client.expire(key, ttlSeconds);
      return value;
    },
    async get(key) {
      return (await client.get<number>(key)) ?? 0;
    },
  };
```

```ts
import { ANON_DAILY_BUDGET_CENTS, ANON_TRIAL_SECONDS } from '@/lib/billing/pricing';
import type { QuotaStore } from './redis';

const TRIAL_TTL_SECONDS = 60 * 60 * 24 * 30;

export type ClaimResult =
  | { granted: true; maxDurationSeconds: number }
  | { granted: false; reason: 'already_used' | 'budget_exhausted' | 'bot_check_failed' };

export function budgetKey(now = new Date()): string {
  return `anon:spend:${now.toISOString().slice(0, 10)}`;
}

export async function claimAnonTrial(input: {
  key: string;
  store: QuotaStore;
  verifyBot: () => Promise<boolean>;
  now?: Date;
}): Promise<ClaimResult> {
  if (!(await input.verifyBot())) {
    return { granted: false, reason: 'bot_check_failed' };
  }

  const spent = await input.store.get(budgetKey(input.now));
  if (spent >= ANON_DAILY_BUDGET_CENTS) {
    return { granted: false, reason: 'budget_exhausted' };
  }

  const claimed = await input.store.claimOnce(`anon:trial:${input.key}`, TRIAL_TTL_SECONDS);
  if (!claimed) return { granted: false, reason: 'already_used' };

  return { granted: true, maxDurationSeconds: ANON_TRIAL_SECONDS };
}

export async function recordAnonSpend(
  store: QuotaStore,
  cents: number,
  now = new Date(),
): Promise<void> {
  await store.incrementBy(budgetKey(now), cents, 60 * 60 * 48);
}
```

```ts
import { NextResponse } from 'next/server';
import { serviceClient } from '@/lib/supabase/server';
import { isSupportedAudio, storagePathFor } from '@/lib/jobs/media';
import { anonKey } from '@/lib/quota/identity';
import { claimAnonTrial } from '@/lib/quota/anon';
import { redisStore } from '@/lib/quota/redis';
import { verifyTurnstile } from '@/lib/turnstile';
import { currentUser } from '@/lib/auth';
import { ANON_TRIAL_SECONDS } from '@/lib/billing/pricing';

export async function POST(request: Request) {
  const body = (await request.json()) as {
    filename: string;
    mimeType: string;
    fingerprint: string;
    turnstileToken: string | null;
  };

  if (!isSupportedAudio(body.mimeType)) {
    return NextResponse.json({ error: 'unsupported_format' }, { status: 415 });
  }

  const ip = request.headers.get('x-forwarded-for')?.split(',')[0].trim() ?? '0.0.0.0';
  const user = await currentUser();
  const db = serviceClient();

  let maxDurationSeconds: number | null = null;
  let anon: string | null = null;

  if (user) {
    const { data: account } = await db
      .from('accounts')
      .select('balance_cents, minutes_included, minutes_used')
      .eq('user_id', user.id)
      .single();

    const hasPlanMinutes =
      (account?.minutes_included ?? 0) > (account?.minutes_used ?? 0);
    if (!hasPlanMinutes && (account?.balance_cents ?? 0) <= 0) {
      return NextResponse.json({ error: 'insufficient_balance' }, { status: 402 });
    }
  } else {
    anon = anonKey(ip, body.fingerprint);
    const claim = await claimAnonTrial({
      key: anon,
      store: redisStore(),
      verifyBot: () => verifyTurnstile(body.turnstileToken, ip),
    });

    if (!claim.granted) {
      return NextResponse.json({ error: claim.reason }, { status: 403 });
    }
    maxDurationSeconds = ANON_TRIAL_SECONDS;
  }

  const { data: job, error } = await db
    .from('jobs')
    .insert({
      user_id: user?.id ?? null,
      anon_key: anon,
      filename: body.filename,
      storage_path: '',
      status: 'awaiting_upload',
    })
    .select('id')
    .single();

  if (error || !job) {
    return NextResponse.json({ error: 'generic' }, { status: 500 });
  }

  const path = storagePathFor(job.id, body.filename);
  await db.from('jobs').update({ storage_path: path }).eq('id', job.id);

  const { data: signed, error: signError } = await db.storage
    .from('audio')
    .createSignedUploadUrl(path);

  if (signError || !signed) {
    return NextResponse.json({ error: 'generic' }, { status: 500 });
  }

  const response = NextResponse.json({
    jobId: job.id,
    uploadUrl: signed.signedUrl,
    uploadToken: signed.token,
    path,
    maxDurationSeconds,
  });

  if (anon) {
    // Lets the anonymous visitor re-open their own result page later.
    response.cookies.set('golos_anon', anon, {
      httpOnly: true,
      sameSite: 'lax',
      secure: process.env.NODE_ENV === 'production',
      maxAge: 60 * 60 * 24 * 30,
    });
  }

  return response;
}
```

```ts
    .from('jobs')
    .update({ provider_job_id: providerJobId, status: 'processing', truncated: isAnonymous })
```

```ts
import { chargeFor, type PlanState } from '@/lib/billing/cost';
import { wordCount } from '@/lib/transcript/turns';
import type { TranscriptionResult } from '@/lib/transcription/types';
import type { Segment } from '@/lib/transcript/types';

export type Completion = {
  job: {
    status: 'done' | 'failed';
    duration_seconds: number;
    language: string | null;
    cost_cents: number;
    error: string | null;
    completed_at: string;
  };
  transcript: {
    job_id: string;
    text: string;
    segments: Segment[];
    word_count: number;
  } | null;
  ledgerEntry: {
    user_id: string;
    delta_cents: number;
    reason: string;
    job_id: string;
    external_id: string;
  } | null;
  minutesUsedDelta: number;
  anonSpendCents: number;
};

/**
 * Pure decision function: given a provider result, decide what to write.
 * Kept separate from the database calls so the money rules are testable.
 */
export function buildCompletion(input: {
  jobId: string;
  userId: string | null;
  result: TranscriptionResult;
  plan: PlanState;
}): Completion {
  const { jobId, userId, result, plan } = input;
  const completedAt = new Date().toISOString();

  if (result.status === 'failed') {
    return {
      job: {
        status: 'failed',
        duration_seconds: 0,
        language: null,
        cost_cents: 0,
        error: result.error,
        completed_at: completedAt,
      },
      transcript: null,
      ledgerEntry: null,
      minutesUsedDelta: 0,
      anonSpendCents: 0,
    };
  }

  const charge = chargeFor(result.durationSeconds, plan);

  return {
    job: {
      status: 'done',
      duration_seconds: result.durationSeconds,
      language: result.language,
      cost_cents: charge.centsFromCredit,
      error: null,
      completed_at: completedAt,
    },
    transcript: {
      job_id: jobId,
      text: result.text,
      segments: result.segments,
      word_count: wordCount(result.text),
    },
    ledgerEntry:
      userId !== null && charge.centsFromCredit > 0
        ? {
            user_id: userId,
            delta_cents: -charge.centsFromCredit,
            reason: 'transcription',
            job_id: jobId,
            // Job-scoped: a replayed webhook collides on the unique index.
            external_id: `job:${jobId}:debit`,
          }
        : null,
    minutesUsedDelta: charge.minutesFromPlan,
    anonSpendCents: userId === null ? charge.centsFromCredit : 0,
  };
}
```

```ts
import { after } from 'next/server';
import { NextResponse } from 'next/server';
import { serviceClient } from '@/lib/supabase/server';
import { transcriptionProvider } from '@/lib/transcription';
import { buildCompletion } from '@/lib/jobs/complete';
import { recordAnonSpend } from '@/lib/quota/anon';
import { redisStore } from '@/lib/quota/redis';

export async function POST(request: Request) {
  const provider = transcriptionProvider();

  if (!provider.verifyWebhook(request.headers)) {
    return NextResponse.json({ error: 'unauthorized' }, { status: 401 });
  }

  const body = (await request.json()) as { transcript_id: string; status: string };

  // Respond immediately — the provider gives us 10 seconds before it retries,
  // and fetching plus storing a two-hour transcript can exceed that.
  after(async () => {
    await applyCompletion(body.transcript_id);
  });

  return NextResponse.json({ ok: true });
}

export async function applyCompletion(providerJobId: string): Promise<void> {
  const db = serviceClient();

  const { data: job } = await db
    .from('jobs')
    .select('id, user_id, status')
    .eq('provider_job_id', providerJobId)
    .single();

  if (!job) return;
  if (job.status === 'done' || job.status === 'failed') return; // already applied

  const result = await transcriptionProvider().fetchResult(providerJobId);

  const { data: account } = job.user_id
    ? await db
        .from('accounts')
        .select('minutes_included, minutes_used')
        .eq('user_id', job.user_id)
        .single()
    : { data: null };

  const completion = buildCompletion({
    jobId: job.id,
    userId: job.user_id,
    result,
    plan: {
      minutesIncluded: account?.minutes_included ?? 0,
      minutesUsed: account?.minutes_used ?? 0,
    },
  });

  if (completion.transcript) {
    await db.from('transcripts').upsert(completion.transcript, { onConflict: 'job_id' });
  }

  if (completion.ledgerEntry) {
    // Duplicate delivery collides on external_id; ignore that one error only.
    const { error } = await db.from('ledger').insert(completion.ledgerEntry);
    if (error && error.code !== '23505') throw error;
  }

  if (completion.minutesUsedDelta > 0 && job.user_id) {
    await db.rpc('add_minutes_used', {
      p_user_id: job.user_id,
      p_minutes: completion.minutesUsedDelta,
    });
  }

  if (completion.anonSpendCents > 0) {
    await recordAnonSpend(redisStore(), completion.anonSpendCents);
  }

  await db.from('jobs').update(completion.job).eq('id', job.id);
}
```

```ts
const GIVE_UP_AFTER_HOURS = 6;

export async function GET(request: Request) {
  if (request.headers.get('authorization') !== `Bearer ${process.env.CRON_SECRET}`) {
    return NextResponse.json({ error: 'unauthorized' }, { status: 401 });
  }

  const db = serviceClient();
  const staleBefore = new Date(Date.now() - STALE_AFTER_MINUTES * 60_000).toISOString();
  const giveUpBefore = new Date(Date.now() - GIVE_UP_AFTER_HOURS * 3_600_000).toISOString();

  const { data: stuck } = await db
    .from('jobs')
    .select('id, provider_job_id, created_at')
    .eq('status', 'processing')
    .lt('created_at', staleBefore)
    .limit(50);

  let recovered = 0;
  let abandoned = 0;

  for (const job of stuck ?? []) {
    if (job.created_at < giveUpBefore) {
      await db
        .from('jobs')
        .update({ status: 'failed', error: 'timed_out', completed_at: new Date().toISOString() })
        .eq('id', job.id);
      abandoned += 1;
      continue;
    }

    if (job.provider_job_id) {
      await applyCompletion(job.provider_job_id);
      recovered += 1;
    }
  }

  return NextResponse.json({ checked: stuck?.length ?? 0, recovered, abandoned });
}
```

```ts
export type UploadProgress = (fraction: number) => void;

/**
 * XMLHttpRequest rather than fetch: fetch still has no upload progress events,
 * and the progress bar is the only honest feedback during a 120 MB upload.
 */
export function uploadToStorage(
  url: string,
  file: File,
  onProgress: UploadProgress,
): Promise<void> {
  return new Promise((resolve, reject) => {
    const xhr = new XMLHttpRequest();
    xhr.open('PUT', url);
    xhr.setRequestHeader('content-type', file.type);

    xhr.upload.onprogress = (event) => {
      if (event.lengthComputable) onProgress(event.loaded / event.total);
    };
    xhr.onload = () =>
      xhr.status >= 200 && xhr.status < 300
        ? resolve()
        : reject(new Error(`upload failed: ${xhr.status}`));
    xhr.onerror = () => reject(new Error('upload failed: network'));

    xhr.send(file);
  });
}
```

```ts
import { NextResponse } from 'next/server';
import { serviceClient } from '@/lib/supabase/server';
import { currentUser } from '@/lib/auth';
import { toTurns } from '@/lib/transcript/turns';
import { toPlainText } from '@/lib/export/txt';
import { toSrt } from '@/lib/export/srt';
import { toDocx } from '@/lib/export/docx';

export async function GET(request: Request, ctx: { params: Promise<{ id: string }> }) {
  const { id } = await ctx.params;
  const format = new URL(request.url).searchParams.get('format') ?? 'txt';

  // Export is the conversion point: viewing is free, taking the file is not.
  const user = await currentUser();
  if (!user) return NextResponse.json({ error: 'sign_in_required' }, { status: 401 });

  const db = serviceClient();
  const { data } = await db
    .from('transcripts')
    .select('segments, jobs!inner(user_id, filename)')
    .eq('job_id', id)
    .single();

  if (!data || data.jobs.user_id !== user.id) {
    return NextResponse.json({ error: 'not_found' }, { status: 404 });
  }

  const turns = toTurns(data.segments);
  const label = (n: number) => `Speaker ${n}`;
  const base = data.jobs.filename.replace(/\.[^.]+$/, '');

  if (format === 'srt') {
    return new NextResponse(toSrt(turns), {
      headers: {
        'content-type': 'application/x-subrip; charset=utf-8',
        'content-disposition': `attachment; filename="${base}.srt"`,
      },
    });
  }

  if (format === 'docx') {
    const buffer = await toDocx(turns, label);
    return new NextResponse(new Uint8Array(buffer), {
      headers: {
        'content-type':
          'application/vnd.openxmlformats-officedocument.wordprocessingml.document',
        'content-disposition': `attachment; filename="${base}.docx"`,
      },
    });
  }

  return new NextResponse(toPlainText(turns, label), {
    headers: {
      'content-type': 'text/plain; charset=utf-8',
      'content-disposition': `attachment; filename="${base}.txt"`,
    },
  });
}
```

```ts
export type PaymentEvent = {

  userId: string;

export type CreditResult = {
  ledgerEntry: {
    user_id: string;
    delta_cents: number;
    reason: string;
    job_id: null;
    external_id: string;
```

```ts
import { NextResponse } from 'next/server';
import { validateEvent, WebhookVerificationError } from '@polar-sh/sdk/webhooks';
import { serviceClient } from '@/lib/supabase/server';
import { creditFromPayment } from '@/lib/billing/credit';

export async function POST(request: Request) {
  const raw = await request.text();

  let event;
  try {
    event = validateEvent(
      raw,
      Object.fromEntries(request.headers),
      process.env.POLAR_WEBHOOK_SECRET!,
    );
  } catch (error) {
    if (error instanceof WebhookVerificationError) {
      return NextResponse.json({ error: 'unauthorized' }, { status: 401 });
    }
    throw error;
  }

  const userId = event.data.metadata?.userId as string | undefined;
  if (!userId) return NextResponse.json({ ok: true });

  const kind = event.type.startsWith('subscription.') ? 'subscription' : 'topup';
  const result = creditFromPayment({
    eventId: event.data.id,
    userId,
    amountCents: event.data.amount ?? 0,
    kind,
  });

  const db = serviceClient();

  if (result.ledgerEntry) {
    const { error } = await db.from('ledger').insert(result.ledgerEntry);
    if (error && error.code !== '23505') throw error; // 23505 = replayed event
  }

  if (result.planUpdate) {
    await db.from('accounts').update(result.planUpdate).eq('user_id', userId);
  }

  return NextResponse.json({ ok: true });
}
```

`userId` travels in the checkout's `metadata` — set it when creating the checkout session so the webhook knows whose account to credit.
